   1| // ============================================================================
   2| // 全局环境配置（与 my_app 部署拓扑一致）—— 单一真相源
   3| // ----------------------------------------------------------------------------
   4| // 设计目标：所有"主机/域名"只在此处定义一处，远程 IP 变更时只需改这里（或页面上
   5| // 的「环境配置」弹窗）即可全站生效。每条看板只存"相对路径 rel"，运行时由
   6| // 主机 + rel 拼出完整 URL，因此已持久化到 localStorage 的旧数据也能自动跟随。
   7| //
   8| // 部署拓扑：
   9| //  - FineBI 生产机：192.168.100.19，nginx 80 端口反代
  10| //      /webroot/  → FineBI 决策平台
  11| //  - Data Insight MVP (my_app)：192.168.100.20
  12| //      /myapp/  → vite 7090（Data Insight MVP 前端，base=/myapp/）
  13| //      /api/    → vite 7090 → uvicorn 7038（my_app 后端）
  14| //  - 门户部署到同一台 nginx 时，API 走同源相对路径 /api/tasks 即可；
  15| //    本地 dev（端口 3000）时通过 vite proxy 将 /api 转发到 my_app 主机。
  16| //
  17| // 优先级：localStorage 覆盖  >  VITE_* 环境变量  >  下方硬编码兜底
  18| // ============================================================================
  19| 
  20| import type { DashboardItem, PlatformType } from './types/dashboard';
  21| 
  22| // ============================================================================
  23| // 门户「自身」后端 API 前缀（/api/auth/*、/api/portal-default）
  24| // ----------------------------------------------------------------------------
  25| // 必须带应用 base 前缀（/uniplam/api），不能再用绝对路径 /api：
  26| //  门户页面挂在 nginx 的 /uniplam/ 子路径下，前端若请求 /api/auth/login，
  27| //  会命中 nginx 的通用 `location /api/` → my_app 后端 7038；7038 上 GET
  28| //  /api/auth/login 是 SPA 兜底静态页（仅允许 GET），POST 就被回成
  29| //  「405 Method Not Allowed」——这正是其他电脑（走 80 端口 nginx）登录失败的原因。
  30| //  带上 base 前缀后，请求路径 /uniplam/api/... 天然落在 `location /uniplam/`
  31| //  （→ 9267 门户自身服务）里，与 my_app 的 /api/ 彻底不冲突。
  32| // 本地直连 http://localhost:9267/uniplam/ 同样走这个前缀，行为一致。
  33| // ============================================================================
  34| export const PORTAL_API = `${(import.meta.env.BASE_URL || '/').replace(/\/+$/, '')}/api`;
  35| 
  36| /** 端点配置：仅主机/域名与 API 基址，路径随各看板 rel 携带 */
  37| export interface EnvConfig {
  38|   /** FineBI 决策平台主机（如 192.168.100.19） */
  39|   finebiHost: string;
  40|   /** Data Insight MVP (my_app) 主机（如 192.168.100.20） */
  41|   myappHost: string;
  42|   /** my_app 后端 API 基址：留空=同源（/api/...），跨域时填 http://192.168.100.20 */
  43|   myappApiBase: string;
  44| }
  45| 
  46| const ENV_STORAGE_KEY = 'noc_portal_env';
  47| 
  48| /** 硬编码兜底（与历史部署一致） */
  49| const FALLBACK_ENV: EnvConfig = {
  50|   finebiHost: '192.168.100.19',
  51|   myappHost: '192.168.100.20',
  52|   myappApiBase: '',
  53| };
  54| 
  55| /** 由 VITE_* 环境变量得到的默认值（环境变量缺失时回退到硬编码） */
  56| function defaultEnv(): EnvConfig {
  57|   return {
  58|     finebiHost:
  59|       (import.meta.env.VITE_FINEBI_HOST as string | undefined) || FALLBACK_ENV.finebiHost,
  60|     myappHost:
  61|       (import.meta.env.VITE_MYAPP_HOST as string | undefined) || FALLBACK_ENV.myappHost,
  62|     myappApiBase:
  63|       (import.meta.env.VITE_MYAPP_API_BASE as string | undefined) || FALLBACK_ENV.myappApiBase,
  64|   };
  65| }
  66| 
  67| // —— 运行时单例（可被页面「环境配置」弹窗改写）——
  68| let currentEnv: EnvConfig = loadEnvConfig();
  69| 
  70| function loadEnvConfig(): EnvConfig {
  71|   try {
  72|     const raw = localStorage.getItem(ENV_STORAGE_KEY);
  73|     if (raw) {
  74|       const parsed = JSON.parse(raw) as Partial<EnvConfig>;
  75|       return { ...defaultEnv(), ...parsed };
  76|     }
  77|   } catch (e) {
  78|     console.error('[config] 解析环境配置失败，使用默认配置', e);
  79|   }
  80|   return defaultEnv();
  81| }
  82| 
  83| /** 读取当前生效的配置副本 */
  84| export function getEnvConfig(): EnvConfig {
  85|   return { ...currentEnv };
  86| }
  87| 
  88| /** 保存配置到 localStorage 并刷新运行时单例 */
  89| export function saveEnvConfig(cfg: EnvConfig): void {
  90|   currentEnv = {
  91|     finebiHost: (cfg.finebiHost || FALLBACK_ENV.finebiHost).trim(),
  92|     myappHost: (cfg.myappHost || FALLBACK_ENV.myappHost).trim(),
  93|     myappApiBase: (cfg.myappApiBase || '').trim(),
  94|   };
  95|   try {
  96|     localStorage.setItem(ENV_STORAGE_KEY, JSON.stringify(currentEnv));
  97|   } catch (e) {
  98|     console.error('[config] 保存环境配置失败', e);
  99|   }
 100| }
 101| 
 102| /** 恢复为默认（清除 localStorage 覆盖） */
 103| export function resetEnvConfig(): EnvConfig {
 104|   currentEnv = defaultEnv();
 105|   try {
 106|     localStorage.removeItem(ENV_STORAGE_KEY);
 107|   } catch (e) {
 108|     console.error('[config] 清除环境配置失败', e);
 109|   }
 110|   return { ...currentEnv };
 111| }

 130| /** Data Insight MVP 前端入口 */
 131| export function getMyappHomeUrl(): string {
 132|   return `http://${currentEnv.myappHost}/myapp/`;
 133| }
 134| 
 135| /**
 136|  * 拼 my_app 深链（含门户鉴权 token）：`?page=<pageKey>[&t=token][#hash]`。
 137|  * token 取自 uniplam 门户登录态（localStorage liberia_noc_auth_v1），my_app 前端
 138|  * 会用 ?t= 换自己的会话 Cookie 并随即从地址栏抽走。未登录（无 token）则不加 t，
 139|  * 保持原链接 —— my_app 会返回 401 引导登录。彩窗/深链统一走这里，勿直接拼字符串。
 140|  */
 141| export function getMyappLink(pageKey: string, hash = ''): string {
 142|   const token = localStorage.getItem('liberia_noc_auth_v1');
 143|   const tokenParam = token ? `&t=${encodeURIComponent(token)}` : '';
 144|   return `${getMyappHomeUrl()}?page=${pageKey}${tokenParam}${hash ? '#' + hash : ''}`;
 145| }
 146| 
 147| /** my_app 任务列表接口（Task Manager 实时状态） */
 148| export function getMyappTasksApi(): string {
 149|   return `${currentEnv.myappApiBase || ''}/api/tasks`;
 150| }

 152| // ============================================================================
 153| // 看板 URL 解析（统一由全局主机配置驱动）
 154| // ============================================================================
 155| 
 156| /**
 157|  * 从完整 URL 提取"相对路径 rel"：去掉 http(s)://host 前缀，保留 path+query+hash。
 158|  * 用于：旧数据迁移、手动编辑看板时自动抽取可跟随主机变化的相对路径。
 159|  */
 160| export function buildRel(platform: PlatformType, url: string): string | undefined {
 161|   if (platform === 'external') return undefined;
 162|   return url.replace(/^https?:\/\/[^/]+/, '');
 163| }
 164| 
 165| /**
 166|  * 解析看板真实访问地址：
 167|  *  - 带 rel 且非外部平台：用全局主机 + rel 实时拼出（改 IP 即时生效）
 168|  *  - 否则退回 item.url（兼容纯外部大屏 / 历史兜底）
 169|  */
 170| export function resolveUrl(item: DashboardItem): string {
 171|   if (item.rel && item.platform !== 'external') {
 172|     const host = item.platform === 'finebi' ? currentEnv.finebiHost : currentEnv.myappHost;
 173|     const base = `http://${host}${item.rel}`;
 174|     if (item.platform !== 'finebi') {
 175|       const token = localStorage.getItem('liberia_noc_auth_v1');
 176|       if (token) {
 177|         return `${base}${base.includes('?') ? '&' : '?'}t=${encodeURIComponent(token)}`;
 178|       }
 179|     }
 180|     return base;
 181|   }
 182|   return item.url;
 183| }
