2148|  /** 后端 generated_at → 浏览器本地 HH:MM:SS。
2149|      2026-09-18：后端 portal_qc.py 的时间戳写死 +08:00（datetime.now(timezone(hours=8))），
2150|      而页面右上角时钟用的是浏览器本地时区（本机 = 利比里亚 UTC）—— 直接切字符串会看到
2151|      「时钟 18:07」旁边写着「更新于 02:07」，像是 8 小时前的旧数据，这也是「QC 好像没在
2152|      自动更新」这类误判的来源之一。这里按浏览器本地时区重算，与页面时钟同一把尺子；
2153|      解析失败（后端换了格式）退回原字符串截取，不显示 Invalid Date。 */
2154|  function qcClock(iso) {
2155|    var d = new Date(iso);
2156|    if (isNaN(d.getTime())) return String(iso).slice(11, 19);
2157|    function p(v) { return String(v).padStart(2, '0'); }
2158|    return p(d.getHours()) + ':' + p(d.getMinutes()) + ':' + p(d.getSeconds());
2159|  }

2161|  /** 距下一个「stepMin 分钟的网格 + offsetSec 秒」还有多少毫秒（2026-09-18 加）。
2162|      两档节奏共用：全量 = 60 分钟网格 + 390s（每小时 :06:30，紧跟 eflow_sync 跑完）；
2163|      告警 = 5 分钟网格 + 30s（:00:30 / :05:30 / …，紧跟 eflow_active_alarm 每 5 分钟的同步）。
2164|
2165|      用**墙上的钟**算，而不是 interval 累加：interval 从进页面那一刻起算，
2166|      进页面时若是 :07，:06 同步好的数据就要再等 59 分钟才上屏。
2167|      网格以「本地当天 0 点」为原点（而不是 epoch），这样 +05:30 这类半小时时区
2168|      也能对齐到本地的 :06，不会整体偏 30 分钟。 */
2169|  function qcNextDelay(stepMin, offsetSec, now) {
2170|    var n = now || new Date();
2171|    var step = stepMin * 60000;
2172|    var dayStart = new Date(n.getFullYear(), n.getMonth(), n.getDate()).getTime();
2173|    var since = n.getTime() - dayStart;                 // 本地当天已过的毫秒
2174|    var target = dayStart + Math.floor(since / step) * step + offsetSec * 1000;
2175|    if (target <= n.getTime()) target += step;
2176|    return target - n.getTime();
2177|  }
