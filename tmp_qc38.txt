 2444:   function qcCell(v) {
 2445:     if (v == null || v === '') return '';
 2446:     var s = String(v);
 2447:     if (/^\d{13}$/.test(s)) {
 2448:       var dt = new Date(Number(s));
 2449:       if (!isNaN(dt.getTime())) {
 2450:         var p = function (n) { return (n < 10 ? '0' : '') + n; };
 2451:         return dt.getFullYear() + '-' + p(dt.getMonth() + 1) + '-' + p(dt.getDate()) +
 2452:           ' ' + p(dt.getHours()) + ':' + p(dt.getMinutes()) + ':' + p(dt.getSeconds());
 2453:       }
 2454:     }
 2455:     return s.replace(/^(-?\d+)\.0+$/, '$1');
 2456:   }
