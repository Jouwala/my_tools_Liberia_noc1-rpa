    # -------------------------------------------------------------------------
    # 访问日志 / 错误日志（2026-09-21 排查「门户页面打开变慢」时补上）
    # -------------------------------------------------------------------------
    # 改动前 access_log / error_log 都是注释状态，走默认值 logs/access.log、
    # logs/error.log。而 conf 和 logs 都在 \\192.168.100.19\e 这个共享上，本机
    # (MSNOC1) 与 .19 两台机器的 nginx 往同一批文件里写：混在一起没法归因，而且
    # access.log 已经涨到 826MB、没人轮转。现在：
    #   · 文件名带 $hostname —— 两台机器各写各的（access-MSNOC1.log 等）。conf 是
    #     两台共用的，所以 .19 那边的 nginx 也要各自 reload 一次才会切到新文件名。
    #   · 格式 timed 里带 rt/urt/uct/uht（总耗时 / 上游耗时 / 连上游耗时 / 上游首字节）。
    #     下次再出现「某个请求慢」，直接从 access log 就能看出慢在哪一段，不用再靠
    #     CDP + curl 一点点试。
    #   · error_log 不支持变量，只能固定文件名，所以仍与 .19 共用一份 error.log。
    #   · 注意：nginx 不允许「带变量的文件名 + buffer=」（实测报
    #     [emerg] buffered logs cannot have variables in name），所以这里没法同时
    #     用 buffer 省掉 SMB 写日志的往返；这一点与改动前一致（原来也是不缓冲），
    #     不算回退。要缓冲就得用固定文件名、放弃按机器分文件。
    # 轮转：仓库根目录 rotate_nginx_logs.ps1 + install_nginx_log_rotate_task.bat
    # -------------------------------------------------------------------------
    log_format  timed  '$remote_addr - $remote_user [$time_local] "$request" '
                       '$status $body_bytes_sent "$http_referer" "$http_user_agent" '
                       'rt=$request_time urt=$upstream_response_time '
                       'uct=$upstream_connect_time uht=$upstream_header_time';
    access_log  logs/access-$hostname.log  timed;
    open_log_file_cache max=100 inactive=60s valid=1m;
    error_log   logs/error.log  warn;

    # -------------------------------------------------------------------------
    # 所有服务收在 80 端口，按「路径」区分：根=导航总览，各服务走子路径
    # -------------------------------------------------------------------------
    # =========================================================================
    # 门户反代：按 Referer 识别「门户自己发出的请求」
    # -------------------------------------------------------------------------
    # 门户页面里用的是根绝对路径（/api/oauth2/...、/uportal/...、/iui/...），
    # 早期方案用 sub_filter 把 "…/api/" 改写成 "…/portal/eflow/api/"，结果门户 JS
    # 会再对字符串做一次拼接，出现 /portal/eflow/uportal/portal/eflow/api/... 双前缀
    # 404。改为：内容一律不改写，nginx 依据 Referer 把门户的请求原样转给上游。
    # 门户与 my_app 同源（都是 192.168.100.20），浏览器默认策略对同源请求发完整
    # Referer，因此可以准确区分「门户发的」与「my_app 发的」/api/ 请求。
    # =========================================================================
    map $http_referer $portal_ref {
        default                                       "";
        "~^https?://[^/]+/portal/eflow/"              "eflow";
        "~^https?://[^/]+/(uportal|iui|framework)/"   "eflow";
    }

    # 门户有跨站访问防护（实测：带会话但来源不符时返回 403
    # "This request is prohibited for cross-site access"），故把 Origin/Referer 改成
    # 门户自己的地址。客户端没发 Origin 时保持为空（凭空加 Origin 反而会触发校验）。
    map $http_origin $portal_origin_rewritten {
        ""        "";
        default   "https://10.100.20.8:28001";
    }

    # =========================================================================
    # B2 安全整改（2026-10-01）：80 端口强制跳 HTTPS
    # -------------------------------------------------------------------------
    # 修复前实测：全站只有 80 明文，https://localhost/ 连接直接失败(000)，
    # 登录口令与 Bearer token 明文穿越客户机房网络。
    #
    # 实现方式（关键，别改成"复制一份 443 server"）：
    #   本文件的服务 location 清单有 200 余行，**复制成两份必然分叉** ——
    #   本文件历史上已因 Y:/D: 两份配置不一致出过线上事故（AGENTS.md 第四节 6b）。
    #   因此这里不复制：同一个 server 块**同时监听 80 与 443**，location 只写一份。
    #   下面这个只含 return 的 server 块（listen 80 default_server）负责把
    #   纯 HTTP 请求 301 到 HTTPS —— nginx 会优先匹配 default_server，
    #   故 80 上的请求全部先跳走，不会走到下面那个双监听块。
    #
    # 证书：自签（certs/portal.crt），SAN 含 192.168.100.20 / .19 / localhost / 127.0.0.1，
    #       有效期至 2036-09-28。浏览器首次访问会提示"不受信任"，这是自签的固有表现；
    #       换客户企业 CA 证书时只替换 certs 下这两个文件，配置无需改动。
    #
    # 回滚：删掉本 server 块 + 下方双监听块里的 443 两行与 ssl_* 指令即可。
    # =========================================================================
    server {
        listen       80 default_server;
        listen       [::]:80 default_server;
        server_name  _;
        return 301 https://$host$request_uri;
    }

        ssl_certificate_key D:/01_RPA/nginx-1.28.0/certs/portal.key;
        ssl_protocols       TLSv1.2 TLSv1.3;

        # IPv6 监听（2026-09-21 补）：Windows 上 localhost 会先解析成 ::1，而 nginx
        # 原来只 listen 80（= 0.0.0.0，纯 IPv4），于是走 localhost 的客户端要先等
        # ::1:80 连接失败（实测 curl 约 2.0s）才回落到 127.0.0.1。补上 IPv6 监听后
        # http://localhost/... 与 http://[::1]/... 直接命中，浏览器之外的工具
        # （curl / 脚本 / Invoke-WebRequest）不再吃这个惩罚。
        # 注：本机 IPv6 只有回环 ::1，够用；.19 已确认有 IPv6（fe80:: 已注册），
        # 所以这条对 .19 也不会导致 nginx 起不来。
        server_name  localhost;

        # 门户登录后主页及其资源的根绝对路径（与 my_app 无冲突，直接转发）
        location /uportal/   { proxy_pass https://10.100.20.8:28001; include portal_eflow_common.conf; }

        # 前缀下的登录页会存成 /portal/eflow_csrftoken，而登录后的主页按无前缀去找，
        # 取不到 token → 所有接口 403 "cross-site access"。故这里按原生路径反代。
        location /api/oauth2/ {
            proxy_pass https://10.100.20.8:28001;
            include    portal_eflow_common.conf;
            sub_filter '</head>' '<script src="/api/portals/inject.js?system=eflow"></script></head>';

        # ---- API 兜底：默认直连本机 my_app 后端（7038）----

            proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;

        # =====================================================================
        # 自托管地图瓦片（利比里亚全境，tilemaker 生成）
        # ---------------------------------------------------------------------
        # 背景：GIS 页原先用 OSM 公共瓦片 + 深色主题靠 CSS invert 滤镜，
        # 既受 OSM usage policy 限制，invert 又把深色公路反成亮白。改为自托管。
        #
        # 数据：D:\tiles-work\liberia.pmtiles（tilemaker v2.4.0 生成，z0–z14，
        # 19,802 瓦片，29.4 MB）。PMTiles 是**单文件**归档，浏览器按 HTTP Range
        # 分段取瓦片，故 nginx 必须正确响应 206 —— 实测 1.28.0 原生支持，
        # 无需任何额外模块（2026-10-02 验证：Range: bytes=0-15 → 206 +
        # Content-Range: bytes 0-15/30874449）。
        #
        # 三个关键点（都踩过或差点踩到）：
        #  1) location 用 `^~` 前缀匹配而非普通前缀。因为文件里已有两条
        #     `location ~* (_bak_|...|\.map$|...)` 正则 location（B5 安全整改），
        #     **正则 location 优先级高于普通前缀 location** —— 若将来瓦片被命名成
        #     liberia_bak_20261002.pmtiles 之类，会被那条规则 404 掉，且现象是
        #     「地图全白」很难联想到安全规则。`^~` 命中后不再尝试正则，彻底规避。
        #  2) alias 末尾与 location 的 /tiles/ 都**必须**带斜杠，否则路径拼接出错。
        #  3) 必须发 Accept-Ranges（nginx 静态文件默认已发）。若将来改成 proxy_pass
        #     到别处，要显式透传 Range 与 206，否则 PMTiles 前端会整包下载或直接失败。
        #
        # CORS：前端若与瓦片不同源（例如 dev 模式 vite:5173 直连）需要放行。
        # 生产走同源 /tiles/ 用不到，但保留可省去将来排查跨域的时间。
        #
        # 回滚：删掉本 location 块即可（配置备份 nginx.conf.bak_20261002_tiles前）。
        # =====================================================================
        location ^~ /tiles/ {
            alias      "D:/01_RPA/nginx-1.28.0/html/tiles/";
            add_header Accept-Ranges bytes always;
            add_header Access-Control-Allow-Origin "*" always;
            add_header Access-Control-Expose-Headers "Content-Range, Content-Length, Accept-Ranges" always;
            types { }                    # 清空 MIME 表，避免 .pmtiles 走到无类型分支
            default_type application/octet-stream;
            # 缓存：实测要点（2026-10-02，nginx 1.28.0）——
            #   a) `expires 30d immutable;` 是**非法**语法，会 emerg: invalid value
            #      （nginx 的 expires 不接受 immutable 参数）；
            #   b) 同时写 `expires` 与 `add_header Cache-Control` 会发出**两条**
            #      Cache-Control 头，nginx 并不会合并它们。
            # 故这里只用 expires 产出唯一一条 Cache-Control: max-age=2592000。
            # 不追加 immutable：瓦片文件名将来若被覆盖（同名重生成），immutable 会让
            # 浏览器 30 天内拒绝重新验证，反而制造"改了不生效"的疑难问题；
            # max-age 已足够，且保留 ETag/Last-Modified 供条件请求。
            expires    30d;
            access_log off;              # 瓦片请求量大，且每个瓦片是一次 Range 请求
        }

        error_page   500 502 503 504  /50x.html;
        location = /50x.html {
            root   html;
        }
    }
