# =============================================================================
# 门户反代（客户端本机登录）— 独立端口架构
#   8081=ZTE UME  8082=CCN UME  8083=iPRAN UME  8084=MW UME  8085=FW UMC  8086=VMAX CXO
#   8087=ShadowBot 占用，勿用；8088=UME iMonitor 大屏（172.17.208.1:28808，见文件末尾；
#        仓库留档 nginx/portal_ume_in_8088.conf）
# 设计要点（与 80 口 eFlow 反代的差异）：
#   * 每个门户独占一个端口 → 会话天然隔离。实测 ZTE/iPRAN/MW 的 Cookie 名与 eFlow
#     完全相同（Z-AUTH-CODE-28001 等），而 Cookie 不分端口，同浏览器多门户共 80 口
#     会互相顶会话，故必须分端口。eFlow 保持占用 80 口原生路径（已验证通过）。
#   * 不剥离任何路径前缀：门户的根绝对路径（/api/oauth2 /uportal /iui …）原样转发。
#   * /api/portals/inject.js 单独代理回本机 7038 → 与登录页同源，注入脚本可取凭据
#     （凭据随请求下发 no-store，不暴露独立明文接口）。
#   * Cookie：domain 改写为访问主机、去 secure（反代为 http）、SameSite=Lax。
#   * 摘 HSTS（否则浏览器对整个 .20 强制 https 搞挂 myapp）与 CSP/XFO。
#   * Origin 置空：同源请求本就不带 Origin，避免跨源探测误判（eFlow 实测结论外推）。
#   * sub_filter 注入自动填表脚本（仅匹配 </head> 一次）。
# =============================================================================

# ---- 8081 ZTE UME 网管 ----
server {
    listen       8081;
    server_name  _;
    location = /api/portals/inject.js {
        proxy_pass http://127.0.0.1:7038;
        proxy_set_header Host $host;
    }
    location / {
        proxy_pass          https://10.100.28.27:28001;
        proxy_ssl_verify    off;
        proxy_ssl_server_name on;
        proxy_http_version  1.1;
        proxy_set_header Host              10.100.28.27:28001;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header Upgrade           $http_upgrade;
        proxy_set_header Connection        "upgrade";
        proxy_set_header Origin            "";
        proxy_set_header Accept-Encoding   "";
        proxy_cookie_domain 10.100.28.27 $host;
        proxy_cookie_flags  ~ nosecure samesite=lax;
        proxy_redirect      https://10.100.28.27:28001/ http://$host:8081/;
        proxy_hide_header   Strict-Transport-Security;
        proxy_hide_header   Content-Security-Policy;
        proxy_hide_header   X-Frame-Options;
        proxy_read_timeout  3600s;
        proxy_send_timeout  3600s;
        sub_filter '</head>' '<script src="/api/portals/inject.js?system=zte_ume"></script></head>';
    }
}

# ---- 8088 UME iMonitor 大屏（guard-imonitor-iui，172.17.208.1:28808）----
# 端口选择：8087 已被 ShadowBot.Shell 占用（127.0.0.1:8087），故用 8088。
# 与上面各门户同构，额外两处必要差异（实测结论）：
#   1) 上游自带 `Content-Security-Policy: frame-ancestors 'self'` → iframe 嵌入被
#      浏览器直接拦死（Chromium: "Framing ... violates ... frame-ancestors 'self'"），
#      故必须摘掉 CSP（连 XFO/HSTS 一起）。
#   2) 登录页内联 frame-bust `top.location = self.location;`：大屏 SPA 无会话时会
#      自己跳到该登录页；若登录页嵌在 myapp 的 iframe 里，这段脚本会把整个 myapp
#      标签页导航走、冲掉 myapp 外壳。用 sub_filter 改成 no-op，登录留在 iframe 内。
#      ⚠️ 门户 JS 里还有两处同类动作（login.min.js 的 `top!=self&&(top.location=self.location)`、
#      security.min.js 登录成功后 `redirectToUrl(top.location, home)`），改 HTML 挡不住，
#      由前端 iframe 的 sandbox（不含 allow-top-navigation）从浏览器层封死 ——
#      见 frontend/src/pages/ImonitorScreen.tsx。两处都需要，别只留一处。
# 仓库留档：nginx/portal_ume_in_8088.conf
server {
    listen       8088;
    server_name  _;
    location = /api/portals/inject.js {
        proxy_pass http://127.0.0.1:7038;
        proxy_set_header Host $host;
    }
    location / {
        proxy_pass          https://172.17.208.1:28808;
        proxy_ssl_verify    off;
        proxy_ssl_server_name on;
        proxy_http_version  1.1;
        proxy_set_header Host              172.17.208.1:28808;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header Upgrade           $http_upgrade;
        proxy_set_header Connection        "upgrade";
        proxy_set_header Origin            "";
        proxy_set_header Accept-Encoding   "";
        proxy_cookie_domain 172.17.208.1 $host;
        proxy_cookie_flags  ~ nosecure samesite=lax;
        proxy_redirect      https://172.17.208.1:28808/ http://$host:8088/;
        proxy_hide_header   Strict-Transport-Security;
        proxy_hide_header   Content-Security-Policy;
        proxy_hide_header   X-Frame-Options;
        proxy_read_timeout  3600s;
        proxy_send_timeout  3600s;
        sub_filter_once off;
        sub_filter 'top.location = self.location;' '/* myapp: frame-bust disabled */';
        sub_filter '</head>' '<script src="/api/portals/inject.js?system=ume_in"></script></head>';
    }
}
